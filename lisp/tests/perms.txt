  (COND ((NULL L) NIL) ((EQ X (CAR L)) (CDR L))\r
        (T (CONS (CAR L) (DEL X (CDR L))))))\r
(DEFUN PUT (X LS)\r
  (COND ((NULL LS) NIL) (T (CONS (CONS X (CAR LS)) (PUT X (CDR LS))))))\r
(DEFUN PERMS (L) (COND ((NULL L) '(NIL)) (T (EACH L))))\r
