(DEFUN APP (A B) (COND (A (CONS (CAR A) (APP (CDR A) B))) (T B)))\r
(DEFUN ROMAN (N) (RO N TB))\r
(DEFUN RO (N L)\r
  (COND ((EQ N 0) NIL) ((< N (CAR (CAR L))) (RO N (CDR L)))\r
        (T (APP (CDR (CAR L)) (RO (- N (CAR (CAR L))) L)))))\r
(DEFUN VL (X L)\r
  (COND ((AND (EQ X (CADR (CAR L))) (NULL (CDR (CDR (CAR L)))))\r
         (CAR (CAR L)))\r
        (T (VL X (CDR L)))))\r

  (COND ((NULL L) 0)\r
        ((AND (CDR L) (< (VL (CAR L) TB) (VL (CADR L) TB)))\r
