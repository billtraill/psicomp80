MAP(A,B,C,D,E) :- NEXT(A,B), NEXT(A,C), NEXT(A,D),\r
  NEXT(B,C), NEXT(B,E), NEXT(C,D), NEXT(C,E).\r
?- MAP(A,B,C,D,E).\r
